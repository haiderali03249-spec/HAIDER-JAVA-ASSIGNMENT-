
import java.util.Scanner;

public class Task11 {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        System.out.print("Enter size of first array: ");
        int n = input.nextInt();
        int[] a = new int[n];

        System.out.println("Enter first sorted array:");
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }

        System.out.print("Enter size of second array: ");
        int m = input.nextInt();
        int[] b = new int[m];

        System.out.println("Enter second sorted array:");
        for (int i = 0; i < m; i++) {
            b[i] = input.nextInt();
        }

        int[] c = new int[n + m];
        int i = 0, j = 0, k = 0;

        while (i < n && j < m) {
            if (a[i] <= b[j]) {
                c[k] = a[i];
                i++;
            } else {
                c[k] = b[j];
                j++;
            }
            k++;
        }

        while (i < n) {
            c[k] = a[i];
            i++;
            k++;
        }

        while (j < m) {
            c[k] = b[j];
            j++;
            k++;
        }

        System.out.println("Third array in sorted order:");
        for (int x = 0; x < c.length; x++) {
            System.out.print(c[x] + " ");
        }

        input.close();
    }
}
